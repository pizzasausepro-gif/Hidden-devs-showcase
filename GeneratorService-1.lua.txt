local Players = game:GetService("Players")
local RunService = game:GetService("RunService")
local CollectionService = game:GetService("CollectionService")

local ProfileStoreManager = require(script.Parent:WaitForChild("ProfileStoreManager", math.huge))
local EconomyService = require(script.Parent:WaitForChild("EconomyService", math.huge))
local Config = require(script.Parent:WaitForChild("Config", math.huge))

local GeneratorService = {}

local GensFolder = nil
while not GensFolder do
	GensFolder = workspace:FindFirstChild("Gens")
	if not GensFolder then task.wait(0.5) end
end

local PlayerFolderAssignments = {}
local ActiveGeneratorLoops = {}
local ActiveConnections = {}
local ClaimCooldowns = {}
local RemoteRateLimits = {}

local function getProfile(player)
	local profile = ProfileStoreManager.GetProfile(player)
	while not profile do
		task.wait(0.1)
		if player.Parent == nil then return nil end
		profile = ProfileStoreManager.GetProfile(player)
	end
	return profile
end

local function getGeneratorRate(genData, genIndex)
	local baseCfg = Config.GENERATORS[genIndex]
	if not baseCfg then return 0 end
	local upgradeMultiplier = 1 + (genData.UpgradeLevel * Config.UPGRADE.MultiplierPerLevel)
	return baseCfg.BaseRate * upgradeMultiplier
end

local function createBillboard(part, genIndex, genName)
	local existing = part:FindFirstChild("EconBillboard")
	if existing then existing:Destroy() end

	local bg = Instance.new("BillboardGui")
	bg.Name = "EconBillboard"
	bg.Size = UDim2.new(0, 200, 0, 80)
	bg.StudsOffset = Vector3.new(0, 4, 0)
	bg.AlwaysOnTop = false
	bg.Adornee = part
	bg.Parent = part

	local title = Instance.new("TextLabel")
	title.Name = "Title"
	title.Size = UDim2.new(1, 0, 0.45, 0)
	title.Position = UDim2.new(0, 0, 0, 0)
	title.BackgroundTransparency = 1
	title.TextColor3 = Color3.fromRGB(255, 220, 50)
	title.TextScaled = true
	title.Font = Enum.Font.GothamBold
	title.Text = genName .. " [Slot " .. genIndex .. "]"
	title.Parent = bg

	local stored = Instance.new("TextLabel")
	stored.Name = "Stored"
	stored.Size = UDim2.new(1, 0, 0.3, 0)
	stored.Position = UDim2.new(0, 0, 0.45, 0)
	stored.BackgroundTransparency = 1
	stored.TextColor3 = Color3.fromRGB(100, 255, 100)
	stored.TextScaled = true
	stored.Font = Enum.Font.Gotham
	stored.Text = "$0 stored"
	stored.Parent = bg

	local rate = Instance.new("TextLabel")
	rate.Name = "Rate"
	rate.Size = UDim2.new(1, 0, 0.25, 0)
	rate.Position = UDim2.new(0, 0, 0.75, 0)
	rate.BackgroundTransparency = 1
	rate.TextColor3 = Color3.fromRGB(180, 180, 255)
	rate.TextScaled = true
	rate.Font = Enum.Font.Gotham
	rate.Text = "Rate: ?"
	rate.Parent = bg

	return bg
end

local function updateBillboard(part, storedMoney, rate)
	local bg = part:FindFirstChild("EconBillboard")
	if not bg then return end
	local storedLabel = bg:FindFirstChild("Stored")
	if storedLabel then storedLabel.Text = "$" .. math.floor(storedMoney) .. " stored" end
	local rateLabel = bg:FindFirstChild("Rate")
	if rateLabel then rateLabel.Text = "$" .. string.format("%.1f", rate) .. "/s" end
end

local function checkRateLimit(player)
	local userId = player.UserId
	local now = tick()
	if not RemoteRateLimits[userId] then
		RemoteRateLimits[userId] = { count = 0, windowStart = now }
	end
	local rl = RemoteRateLimits[userId]
	if now - rl.windowStart > Config.REMOTE_RATE_WINDOW then
		rl.count = 0
		rl.windowStart = now
	end
	rl.count = rl.count + 1
	return rl.count <= Config.REMOTE_RATE_LIMIT
end

function GeneratorService.AssignFolder(player)
	for _, folderName in ipairs(Config.PLAYER_FOLDERS) do
		local inUse = false
		for _, assignment in pairs(PlayerFolderAssignments) do
			if assignment == folderName then
				inUse = true
				break
			end
		end
		if not inUse then
			PlayerFolderAssignments[player.UserId] = folderName
			return folderName
		end
	end
	return nil
end

function GeneratorService.GetAssignedFolder(player)
	local folderName = PlayerFolderAssignments[player.UserId]
	if not folderName then return nil end
	return GensFolder:FindFirstChild(folderName)
end

function GeneratorService.InitGenerators(player)
	local profile = getProfile(player)
	if not profile then return end

	local folder = GeneratorService.GetAssignedFolder(player)
	if not folder then
		warn("[GeneratorService] No folder assigned for " .. player.Name)
		return
	end

	local conns = {}
	ActiveConnections[player.UserId] = conns

	local prestigeMultiplier = profile.Data.PrestigeMultiplier or 1.0

	for genIndex = 1, Config.MAX_GENERATORS_PER_PLAYER do
		local genData = profile.Data.Generators[genIndex]
		local baseCfg = Config.GENERATORS[genIndex]
		if not genData or not baseCfg then continue end

		local part = folder:FindFirstChild("Gen" .. genIndex)
		if not part then continue end

		if genData.Owned then
			local billboard = createBillboard(part, genIndex, baseCfg.Name)
			local rate = getGeneratorRate(genData, genIndex)
			updateBillboard(part, genData.StoredMoney, rate * prestigeMultiplier)

			local conn = part.Touched:Connect(function(hit)
				local char = hit.Parent
				local touchPlayer = Players:GetPlayerFromCharacter(char)
				if not touchPlayer or touchPlayer ~= player then return end

				if not checkRateLimit(touchPlayer) then return end

				local now = tick()
				local cdKey = player.UserId .. "_" .. genIndex
				if ClaimCooldowns[cdKey] and now - ClaimCooldowns[cdKey] < Config.CLAIM_COOLDOWN then return end
				ClaimCooldowns[cdKey] = now

				GeneratorService.ClaimGenerator(touchPlayer, genIndex)
			end)
			table.insert(conns, conn)
		else
			local bg = createBillboard(part, genIndex, baseCfg.Name .. " [LOCKED]")
		end
	end

	local loopThread = task.spawn(function()
		local billboardTimer = 0

		while true do
			local dt = task.wait(Config.GENERATION_TICK_RATE)
			if player.Parent == nil then break end

			local currentProfile = ProfileStoreManager.GetProfile(player)
			if not currentProfile then break end

			local prestige = currentProfile.Data.PrestigeMultiplier or 1.0
			billboardTimer = billboardTimer + dt

			for genIndex = 1, Config.MAX_GENERATORS_PER_PLAYER do
				local gd = currentProfile.Data.Generators[genIndex]
				local baseCfg = Config.GENERATORS[genIndex]
				if not gd or not baseCfg or not gd.Owned then continue end

				local rate = getGeneratorRate(gd, genIndex) * prestige
				gd.StoredMoney = math.min(gd.StoredMoney + rate * dt, baseCfg.MaxStorage)

				if billboardTimer >= Config.BILLBOARD_UPDATE_RATE then
					local folder2 = GeneratorService.GetAssignedFolder(player)
					if folder2 then
						local part = folder2:FindFirstChild("Gen" .. genIndex)
						if part then
							updateBillboard(part, gd.StoredMoney, rate)
						end
					end
				end
			end

			if billboardTimer >= Config.BILLBOARD_UPDATE_RATE then
				billboardTimer = 0
			end
		end
	end)

	ActiveGeneratorLoops[player.UserId] = loopThread
end

function GeneratorService.ClaimGenerator(player, genIndex)
	if type(genIndex) ~= "number" then return false end
	genIndex = math.floor(genIndex)
	if genIndex < 1 or genIndex > Config.MAX_GENERATORS_PER_PLAYER then return false end

	local profile = ProfileStoreManager.GetProfile(player)
	if not profile then return false end

	local genData = profile.Data.Generators[genIndex]
	if not genData then return false end
	if not genData.Owned then return false end

	local folder = GeneratorService.GetAssignedFolder(player)
	if not folder then return false end

	local expectedFolder = PlayerFolderAssignments[player.UserId]
	if not expectedFolder then return false end

	local amount = math.floor(genData.StoredMoney)
	if amount <= 0 then return false end

	genData.StoredMoney = 0
	EconomyService.AddMoney(player, amount)

	return true
end

function GeneratorService.GetGeneratorData(player, genIndex)
	local profile = ProfileStoreManager.GetProfile(player)
	if not profile then return nil end
	return profile.Data.Generators[genIndex]
end

function GeneratorService.Cleanup(player)
	local conns = ActiveConnections[player.UserId]
	if conns then
		for _, c in ipairs(conns) do
			pcall(function() c:Disconnect() end)
		end
		ActiveConnections[player.UserId] = nil
	end

	if ActiveGeneratorLoops[player.UserId] then
		task.cancel(ActiveGeneratorLoops[player.UserId])
		ActiveGeneratorLoops[player.UserId] = nil
	end

	local folder = GeneratorService.GetAssignedFolder(player)
	if folder then
		for genIndex = 1, Config.MAX_GENERATORS_PER_PLAYER do
			local part = folder:FindFirstChild("Gen" .. genIndex)
			if part then
				local bg = part:FindFirstChild("EconBillboard")
				if bg then bg:Destroy() end
			end
		end
	end

	PlayerFolderAssignments[player.UserId] = nil
	ClaimCooldowns[player.UserId .. "_1"] = nil
	ClaimCooldowns[player.UserId .. "_2"] = nil
	ClaimCooldowns[player.UserId .. "_3"] = nil
	ClaimCooldowns[player.UserId .. "_4"] = nil
	RemoteRateLimits[player.UserId] = nil
end

return GeneratorService
