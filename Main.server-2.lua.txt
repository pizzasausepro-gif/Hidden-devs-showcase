local Players = game:GetService("Players")
local ReplicatedStorage = game:GetService("ReplicatedStorage")
local ServerScriptService = game:GetService("ServerScriptService")

local Config = require(ServerScriptService:WaitForChild("Config", math.huge))
local ProfileStoreManager = require(ServerScriptService:WaitForChild("ProfileStoreManager", math.huge))
local EconomyService = require(ServerScriptService:WaitForChild("EconomyService", math.huge))
local GeneratorService = require(ServerScriptService:WaitForChild("GeneratorService", math.huge))
local OfflineIncomeService = require(ServerScriptService:WaitForChild("OfflineIncomeService", math.huge))
local UpgradeService = require(ServerScriptService:WaitForChild("UpgradeService", math.huge))

local EconomyRemotes = ReplicatedStorage:WaitForChild("EconomyRemotes", math.huge)
local RemoteClaim = EconomyRemotes:WaitForChild("ClaimGenerator", math.huge)
local RemoteUpgrade = EconomyRemotes:WaitForChild("PurchaseUpgrade", math.huge)
local RemoteOffline = EconomyRemotes:WaitForChild("OfflineEarnings", math.huge)
local RemoteGetUpgradeInfo = EconomyRemotes:WaitForChild("GetUpgradeInfo", math.huge)
local RemoteGetGenData = EconomyRemotes:WaitForChild("GetGeneratorData", math.huge)

local RemoteRateLimits = {}
local RATE_LIMIT_MAX = Config.REMOTE_RATE_LIMIT
local RATE_WINDOW = Config.REMOTE_RATE_WINDOW

local function checkRateLimit(player)
	local userId = player.UserId
	local now = tick()
	if not RemoteRateLimits[userId] then
		RemoteRateLimits[userId] = { count = 0, windowStart = now }
	end
	local rl = RemoteRateLimits[userId]
	if now - rl.windowStart > RATE_WINDOW then
		rl.count = 0
		rl.windowStart = now
	end
	rl.count = rl.count + 1
	return rl.count <= RATE_LIMIT_MAX
end

local function onPlayerAdded(player)
	task.spawn(function()
		if player.Parent == nil then return end

		local folderName = GeneratorService.AssignFolder(player)
		if not folderName then
			warn("[Bootstrap] No available folder for " .. player.Name .. ". Kicking.")
			player:Kick("Server is full. Please try again later.")
			return
		end

		local profile = ProfileStoreManager.LoadProfile(player)
		if not profile then return end

		EconomyService.InitLeaderstats(player)

		local offlineEarned = OfflineIncomeService.Calculate(player)

		GeneratorService.InitGenerators(player)

		if offlineEarned > 0 then
			task.wait(2)
			if player.Parent ~= nil then
				RemoteOffline:FireClient(player, offlineEarned)
			end
		end
	end)
end

local function onPlayerRemoving(player)
	OfflineIncomeService.SetLastOnline(player)
	GeneratorService.Cleanup(player)
	ProfileStoreManager.SaveProfile(player)
	ProfileStoreManager.ReleaseProfile(player)
	RemoteRateLimits[player.UserId] = nil
end

RemoteClaim.OnServerEvent:Connect(function(player, genIndex)
	if not checkRateLimit(player) then return end
	if type(genIndex) ~= "number" then return end
	GeneratorService.ClaimGenerator(player, genIndex)
end)

RemoteUpgrade.OnServerEvent:Connect(function(player, genIndex)
	if not checkRateLimit(player) then return end
	if type(genIndex) ~= "number" then return end

	local profile = ProfileStoreManager.GetProfile(player)
	if not profile then return end

	local genData = profile.Data.Generators[math.floor(genIndex)]
	if not genData or not genData.Owned then return end

	local success, result = UpgradeService.PurchaseUpgrade(player, genIndex)
	if success then
		GeneratorService.Cleanup(player)
		GeneratorService.InitGenerators(player)
	end
end)

RemoteGetUpgradeInfo.OnServerInvoke = function(player, genIndex)
	if not checkRateLimit(player) then return nil end
	if type(genIndex) ~= "number" then return nil end
	return UpgradeService.GetInfo(player, genIndex)
end

RemoteGetGenData.OnServerInvoke = function(player, genIndex)
	if not checkRateLimit(player) then return nil end
	if type(genIndex) ~= "number" then return nil end

	local profile = ProfileStoreManager.GetProfile(player)
	if not profile then return nil end

	local genData = profile.Data.Generators[math.floor(genIndex)]
	if not genData then return nil end

	local baseCfg = Config.GENERATORS[math.floor(genIndex)]
	return {
		Owned = genData.Owned,
		StoredMoney = math.floor(genData.StoredMoney),
		UpgradeLevel = genData.UpgradeLevel,
		MaxStorage = baseCfg and baseCfg.MaxStorage or 0,
		Name = baseCfg and baseCfg.Name or "Unknown",
	}
end

Players.PlayerAdded:Connect(onPlayerAdded)
Players.PlayerRemoving:Connect(onPlayerRemoving)

for _, player in ipairs(Players:GetPlayers()) do
	task.spawn(onPlayerAdded, player)
end
